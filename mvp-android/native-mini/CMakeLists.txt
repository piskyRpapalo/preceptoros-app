cmake_minimum_required(VERSION 3.22)
project(preceptor_mini LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT LLAMA_CPP_SRC OR NOT EXISTS "${LLAMA_CPP_SRC}/include/llama.h")
  message(FATAL_ERROR "LLAMA_CPP_SRC must point to a verified llama.cpp checkout")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_SRC}" "${CMAKE_BINARY_DIR}/llama")
add_library(mini_llama SHARED mini_jni.cpp)
target_include_directories(mini_llama PRIVATE "${LLAMA_CPP_SRC}/include")
target_link_libraries(mini_llama PRIVATE llama log android)
